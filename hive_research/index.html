🚀 Quick Start

1. Go to Import → enter arXiv ID (e.g. 1706.03762) → click Add
2. Pipeline runs: fetch → download PDF → extract text → LLM analysis → populate graph → index for RAG
3. Visit Graph to see the paper and its linked concepts
4. Ask questions in Chat using RAG over all papers

# --- Setup ---
import requests, json
BASE = "http://localhost:7777"
# --- Add a paper ---
r = requests.post(f"{BASE}/api/add",
json={"id": "1706.03762", "model": "fast"})
print(r.json()) # {status, paper_id, concepts, tags, ...}
# --- Query the graph ---
r = requests.get(f"{BASE}/api/graph")
graph = r.json()
print(f"Papers: {len(graph['nodes'])}, Edges: {len(graph['links'])}")
# --- RAG question ---
r = requests.post(f"{BASE}/api/query",
json={"question": "What methods are used for graph classification?"})
print(r.json()["answer"])
# --- Search arXiv (no import) ---
r = requests.post(f"{BASE}/api/search",
json={"query": "transformer attention"})
for p in r.json()["results"]:
print(p["title"], p["arxiv_id"])
# --- Import batch ---
r = requests.post(f"{BASE}/api/import",
json={"query": "graph neural networks", "model": "fast"})
print(r.json()["added"], "papers added")
# --- Similarity matrix ---
r = requests.post(f"{BASE}/api/similarity",
json={"algorithm": "combined"})
top = sorted(r.json(), key=lambda x: x["score"], reverse=True)[:5]
for s in top: print(f"{s['source_title'][:40]:40} {s['score']:.2f}")
# --- System stats ---
r = requests.get(f"{BASE}/api/stats")
print(r.json()) # {papers, concepts, relations, rag, gpu}
# %% [markdown]
# Hive Research GPU — Jupyter Notebook Guide
# %%
import requests, json, pandas as pd
import matplotlib.pyplot as plt
from IPython.display import Markdown, display
BASE = "http://localhost:7777"
# %% [markdown]
# ## 1. Add Papers to the Knowledge Base
# %%
# Add a single paper by arXiv ID
r = requests.post(f"{BASE}/api/add", json={"id": "1706.03762"})
result = r.json()
display(Markdown(f"**Status:** {result['status']} | **Concepts:** {result.get('concepts',0)}"))
# %% [markdown]
# ## 2. Browse the Knowledge Graph
# %%
# Fetch graph and visualize node types
graph = requests.get(f"{BASE}/api/graph").json()
nodes = graph["nodes"]
types = {}
for n in nodes:
t = n.get("type", "paper")
types[t] = types.get(t, 0) + 1
pd.Series(types).plot(kind="bar", color="#60a5fa")
plt.title("Knowledge Graph Node Types")
plt.show()
# %% [markdown]
# ## 3. RAG Question Answering
# %%
question = "What are the key contributions of the attention mechanism?"
r = requests.post(f"{BASE}/api/query", json={"question": question})
answer = r.json()["answer"]
display(Markdown(f"### Answer\n\n{answer}"))
# %% [markdown]
# ## 4. Paper Similarity Analysis
# %%
sim = requests.post(f"{BASE}/api/similarity",
json={"algorithm": "vector"}).json()
df = pd.DataFrame(sim).sort_values("score", ascending=False).head(10)
df[["source_title", "target_title", "score"]]
# %% [markdown]
# ## 5. Research Pool Monitoring
# %%
pool = requests.get(f"{BASE}/api/pool").json()
for topic, papers in pool.items():
print(f"\n=== {topic} ({len(papers)} papers) ===")
for p in papers[:3]:
print(f" - {p['title'][:60]}...")
# %% [markdown]
# ## 6. System Health Check
# %%
stats = requests.get(f"{BASE}/api/stats").json()
gpu = stats.get("gpu", {})
display(Markdown(f"""
| Metric | Value |
|--------|-------|
| Papers | {stats['papers']} |
| Concepts | {stats['concepts']} |
| RAG Chunks | {stats['rag']['chunks']} |
| GPUs | {gpu.get('device_count', 0)} |
"""))
